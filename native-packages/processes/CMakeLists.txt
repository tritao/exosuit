cmake_minimum_required(VERSION 3.21)
project(processkit LANGUAGES C)

add_library(processkit SHARED src/processkit.c)
target_compile_features(processkit PRIVATE c_std_11)
target_include_directories(processkit PUBLIC include)
set_target_properties(processkit PROPERTIES C_VISIBILITY_PRESET hidden)
if(WIN32)
    target_link_libraries(processkit PRIVATE user32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(processkit PRIVATE Threads::Threads)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(processkit_contract tests/contract.c)
        target_link_libraries(processkit_contract PRIVATE processkit)
        add_test(NAME processkit_contract COMMAND processkit_contract)
        if(WIN32)
            # The contract test loads processkit.dll from its own directory.
            set_target_properties(processkit processkit_contract PROPERTIES
                RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
        endif()
    endif()
endif()
