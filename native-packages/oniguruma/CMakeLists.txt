cmake_minimum_required(VERSION 3.21)
project(textmate_regex LANGUAGES C)

# Oniguruma is compiled into this toolkit on desktop and into the Emscripten
# host for browser builds. Keep its options local to this subdirectory.
cmake_policy(PUSH)
cmake_policy(SET CMP0077 NEW)
set(_textmate_saved_shared ${BUILD_SHARED_LIBS})
set(_textmate_had_shared_cache FALSE)
if(DEFINED CACHE{BUILD_SHARED_LIBS})
    set(_textmate_had_shared_cache TRUE)
    get_property(_textmate_saved_shared_cache CACHE BUILD_SHARED_LIBS PROPERTY VALUE)
endif()
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build Oniguruma statically for TextMate" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(INSTALL_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(INSTALL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_POSIX_API OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/oniguruma oniguruma)
set(BUILD_SHARED_LIBS ${_textmate_saved_shared})
if(_textmate_had_shared_cache)
    set(BUILD_SHARED_LIBS ${_textmate_saved_shared_cache} CACHE BOOL "" FORCE)
else()
    unset(BUILD_SHARED_LIBS CACHE)
endif()
unset(CMAKE_POLICY_DEFAULT_CMP0077)
cmake_policy(POP)
set_target_properties(onig PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(EMSCRIPTEN)
    set(TEXTMATE_REGEX_LIBRARY_TYPE STATIC)
else()
    set(TEXTMATE_REGEX_LIBRARY_TYPE SHARED)
endif()
add_library(textmate_regex ${TEXTMATE_REGEX_LIBRARY_TYPE} src/textmate_regex.c)
target_compile_features(textmate_regex PRIVATE c_std_11)
target_compile_definitions(textmate_regex PRIVATE TEXTMATE_REGEX_BUILDING=1)
target_include_directories(textmate_regex PUBLIC include PRIVATE vendor/oniguruma/src)
target_link_libraries(textmate_regex PRIVATE onig)
set_target_properties(textmate_regex PROPERTIES C_VISIBILITY_PRESET hidden)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(textmate_regex_contract tests/contract.c)
        target_link_libraries(textmate_regex_contract PRIVATE textmate_regex)
        target_compile_features(textmate_regex_contract PRIVATE c_std_11)
        add_test(NAME textmate_regex_contract COMMAND textmate_regex_contract)
    endif()
endif()
